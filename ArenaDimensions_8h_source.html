49inline ArenaDimensions calculate_arena_dimensions(const LevelData& level) {
50 const auto& layout = level.brick_layout;
51 const auto& num_rows = level.bricks.size();
52 const auto& num_cols = level.bricks[0].size();
53
54 // Paddle Gameplay Space at the lower part
55 const float paddle_space = level.paddle_offset.y + level.paddle_height;
56 const float ball_space = level.ball_start_offset.y + (level.ball_radius * 2);
57 const float buffer = paddle_space + ball_space + level.wall_thickness + 4.0F; //extra padding
58
59 const float arena_width = (static_cast<float>(num_cols) * layout.brick_size.x) +
60 ((static_cast<float>(num_cols) - 1) * layout.brick_margin.x) + level.inner_left_margin
61 + level.inner_right_margin;
62
63 const float arena_height = buffer + (static_cast<float>(num_rows) * layout.brick_size.y) +
64 ((static_cast<float>(num_rows) - 1) * layout.brick_margin.y) + level.inner_top_margin
65 + level.inner_bottom_margin;
66
67 const float window_width = arena_width + level.outer_left_margin + level.outer_right_margin
68 + (2.0F * level.wall_thickness);
69 const float window_height = arena_height + level.outer_top_margin + level.outer_bottom_margin
70 + (2.0F * level.wall_thickness);
71
72 return ArenaDimensions{
73 Vec2{arena_width, arena_height},
74 Vec2{window_width, window_height}
75 };
