90enum class CollisionBehavior {
91 None = 0,
92 Clamp = 1 << 0,
93 Bounce = 1 << 1,
94 Destroy = 1 << 2,
95 Trigger = 1 << 3
96};

98inline CollisionBehavior operator|(CollisionBehavior lhs, CollisionBehavior rhs) {
99 return static_cast<CollisionBehavior>(static_cast<int>(lhs) | static_cast<int>(rhs));
100}
101
102inline CollisionBehavior operator&(CollisionBehavior lhs, CollisionBehavior rhs) {
103 return static_cast<CollisionBehavior>(static_cast<int>(lhs) & static_cast<int>(rhs));
